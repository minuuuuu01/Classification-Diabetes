import pandas as pd
import seaborn as sns
import matplotlib.pyplot as plt
from sklearn.model_selection import train_test_split

# 데이터 로드
df = pd.read_csv("diabetes_prediction_dataset.csv", encoding="cp949") 
# ----------------------------------------------------------------------

df.info() # 각 컬럼별 데이터 타입, 결측치 확인 

print(df.describe()) # 수치형 변수에 대한 기초 통계량 확인 

# ------------------------------------------------------------------
# 이상치가 있을법한 컬럼만 추려서 사분위수 사용해서 정상 범위 구한 후, 
# 이상치 범위에 해당되는 데이터 개수와 그중 당뇨에 해당하는 데이터 개수 및 비율 출력

# 분석할 컬럼 리스트
cols = ['BMI 지수', '혈당 수치', '당화혈색소 수치']

for col in cols:
    # 1) 사분위수 계산 및 이상치 범위 설정
    Q1 = df[col].quantile(0.25)
    Q3 = df[col].quantile(0.75)
    IQR = Q3 - Q1
    
    lower_bound = Q1 - 1.5 * IQR
    upper_bound = Q3 + 1.5 * IQR
    
    # 2) 정상 범위 밖 데이터 추출
    outliers = df[(df[col] < lower_bound) | (df[col] > upper_bound)]
    num_outliers = len(outliers)
    
    # 3) 이상치 중 당뇨병 환자(1) 데이터 추출
    outliers_diabetes = outliers[outliers['당뇨병 여부'] == 1]
    num_outliers_diabetes = len(outliers_diabetes)
    
    # 결과 출력
    print(f"[{col} 이상치 분석]")
    print(f"- 정상 범위: {lower_bound:.2f} ~ {upper_bound:.2f}")
    print(f"- 전체 이상치 개수: {num_outliers}개")
    print(f"- 이상치 중 당뇨병 환자 수: {num_outliers_diabetes}개")
    
    if num_outliers > 0:
        proportion = (num_outliers_diabetes / num_outliers) * 100
        print(f"- 이상치 내 당뇨병 환자 비율: {proportion:.2f}%")
    print("-" * 40)

# 'BMI 지수' 컬럼에 대해서만 이상치 처리. 이상치를 전부 상한선(38.5)으로 대체.
# 1. BMI 지수의 사분위수(Q1, Q3) 및 IQR 계산
Q1 = df['BMI 지수'].quantile(0.25)
Q3 = df['BMI 지수'].quantile(0.75)
IQR = Q3 - Q1

# 2. 상한선(Upper Bound) 설정 (하한선은 계산상 14.71이나, 굳이 대치할 필요가 없으므로 상한선만 적용)
upper_limit = Q3 + 1.5 * IQR

# 3. 상한선을 초과하는 값을 상한선으로 대체 (clip 함수 사용)
# clip(upper=값)은 지정한 값보다 큰 데이터를 모두 해당 값으로 고정해줍니다.
df['BMI 지수'] = df['BMI 지수'].clip(upper=upper_limit)

# -----------------------------------------------------------------------
# 범주형 변수 인코딩 
# '성별' 컬럼 인코딩: 레이블 인코딩 방식 
gender_map = {'Female': 0, 'Male': 1, 'Other': 2}
df['성별'] = df['성별'].map(gender_map)

# '흡연 경험' 컬럼 인코딩: 원-핫 인코딩 방식 
df = pd.get_dummies(df, columns=['흡연 경험'], prefix='흡연')

# -------------------------------------------------------------------------
# 데이터 분할(7:2:1)
X = df.drop('당뇨병 여부', axis=1)
y = df['당뇨병 여부']

# 1. 먼저 전체 데이터를 [학습+검증(90%)] : [테스트(10%)]로 나눕니다.
X_temp, X_test, y_temp, y_test = train_test_split(
    X, y, test_size=0.1, random_state=42, stratify=y
)

# 2. 남은 [학습+검증(90%)] 데이터를 다시 [학습(70%)] : [검증(20%)] 비율로 나눕니다.
# (전체 대비 20%를 가져오기 위해 test_size를 약 0.22로 설정합니다: 0.9 * 0.22 ≒ 0.2)
X_train, X_val, y_train, y_val = train_test_split(
    X_temp, y_temp, test_size=0.2222, random_state=42, stratify=y_temp
)
